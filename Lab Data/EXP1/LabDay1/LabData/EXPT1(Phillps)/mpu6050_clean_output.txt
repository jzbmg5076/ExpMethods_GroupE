#include<Wire.h>
const int MPU_addr=0x68; // I2C address of the MPU-6050

int16_t AcX[2050];
int16_t AcY[2050];
int16_t AcZ[2050];
unsigned long start_time;
unsigned long end_time;
int16_t num_cap = 1;

void setup(){
  Wire.begin();
  Wire.beginTransmission(MPU_addr);
  Wire.write(0x6B); // PWR_MGMT_1 register
  Wire.write(0); // set to zero (wakes up the MPU-6050)
  Wire.endTransmission(true);
  Serial.begin(115200); // 9600
  delay(1000);
}

void loop() {
  int i;
  if(num_cap) {
    Serial.println();
    Serial.println("======================================");
    Serial.println("  Starting swing capture...");
    delay(2000);
    start_time = millis();
    for (i=0;i<2048;i++) {
      Wire.beginTransmission(MPU_addr);
      Wire.write(0x3B); // starting with register 0x3B (ACCEL_XOUT_H)
      Wire.endTransmission(false);
      Wire.requestFrom(MPU_addr,6,true); // request a total of 14 registers
      AcX[i]=Wire.read()<<8|Wire.read(); // 0x3B (ACCEL_XOUT_H) & 0x3C (ACCEL_XOUT_L)
      AcY[i]=Wire.read()<<8|Wire.read(); // 0x3D (ACCEL_YOUT_H) & 0x3E (ACCEL_YOUT_L)
      AcZ[i]=Wire.read()<<8|Wire.read(); // 0x3F (ACCEL_ZOUT_H) & 0x40 (ACCEL_ZOUT_L)
    }
    end_time = millis();

    Serial.print("  Samples: 2048   Duration: ");
    Serial.print(end_time-start_time);
    Serial.println(" mS");
    Serial.println("======================================");
    Serial.println();

    Serial.println("% -=- X, Y, Z -=-");

    Serial.print("X = [");
    for (i=0; i<2048; i++) {
      Serial.print(AcX[i]);
      if (i < 2047) Serial.print(", ");
    }
    Serial.println("];");

    Serial.print("Y = [");
    for (i=0; i<2048; i++) {
      Serial.print(AcY[i]);
      if (i < 2047) Serial.print(", ");
    }
    Serial.println("];");

    Serial.print("Z = [");
    for (i=0; i<2048; i++) {
      Serial.print(AcZ[i]);
      if (i < 2047) Serial.print(", ");
    }
    Serial.println("];");

    Serial.println();
    Serial.println("% -=-=- End of capture -=-=-");
    Serial.println("======================================");
    Serial.println();

    num_cap--;
  } else { delay(1000); }
}
